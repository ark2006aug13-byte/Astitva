⚡ Supabase SQL Setup (1-Click Table Creation)

Open your Supabase Dashboard ➔ SQL Editor ➔ paste this SQL script and click Run. This will automatically create the user_data table and allow public submissions. 

-- 1. Create User Data Table
create table if not exists public.user_data (
  id uuid default gen_random_uuid() primary key,
  full_name text not null,
  email text not null,
  phone text,
  target_role text,
  skills text,
  notes text,
  resume_file_name text,
  resume_url text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Enable Row Level Security and Allow Public Inserts
alter table public.user_data enable row level security;

create policy "Allow public inserts" on public.user_data 
  for insert with check (true);

create policy "Allow public selects" on public.user_data 
  for select using (true);

-- 3. Create Storage Bucket for Resumes
insert into storage.buckets (id, name, public) 
values ('resumes', 'resumes', true)
on conflict (id) do nothing;

create policy "Public Access to Resumes" on storage.objects 
  for select using (bucket_id = 'resumes');

create policy "Public Upload to Resumes" on storage.objects 
  for insert with check (bucket_id = 'resumes');
